// ★ 접속로그 (log.html - 관리자 전용 열람현황 분석)
//   기존 action.gs 파일 아무 곳에나 (파일 맨 끝 권장) 통째로 붙여넣고,
//   아래 "doGet / doPost 라우팅 추가" 부분만 기존 doGet/doPost 함수 안에 한 줄씩 추가하세요.
// ================================================================

const VIEWLOG_SHEET_NAME = '접속로그';
const VIEWLOG_HEADERS    = ['시각', '로그인유형', '팀명', '사용자명', 'IP', '페이지'];
const VIEWLOG_MAX_ROWS   = 5000; // getViewLogs가 반환하는 최근 로그 상한 (시트 자체는 계속 누적됨)

function ensureViewLogSheet() {
  const ss    = SpreadsheetApp.getActiveSpreadsheet();
  let   sheet = ss.getSheetByName(VIEWLOG_SHEET_NAME);
  if (!sheet) {
    sheet = ss.insertSheet(VIEWLOG_SHEET_NAME);
    sheet.getRange(1, 1, 1, VIEWLOG_HEADERS.length).setValues([VIEWLOG_HEADERS]);
    sheet.getRange(1, 1, 1, VIEWLOG_HEADERS.length).setBackground('#0f1f3d').setFontColor('#ffffff').setFontWeight('bold');
    sheet.setFrozenRows(1);
    const widths = [140, 90, 100, 100, 120, 260];
    widths.forEach((w, i) => sheet.setColumnWidth(i + 1, w));
  }
  return sheet;
}

function logPageView(body) {
  try {
    const sheet = ensureViewLogSheet();
    const now   = Utilities.formatDate(new Date(), 'Asia/Seoul', 'yyyy-MM-dd HH:mm:ss');
    sheet.appendRow([
      now,
      String(body.loginType || ''),
      String(body.team || ''),
      String(body.userName || ''),
      String(body.ip || ''),
      String(body.page || '')
    ]);
    return respond({ ok: true });
  } catch(err) { return respond({ ok: false, msg: err.toString() }); }
}

function getViewLogs() {
  try {
    const sheet   = ensureViewLogSheet();
    const lastRow = sheet.getLastRow();
    if (lastRow <= 1) return respond({ ok: true, rows: [] });
    const startRow = Math.max(2, lastRow - VIEWLOG_MAX_ROWS + 1);
    const numRows  = lastRow - startRow + 1;
    const data = sheet.getRange(startRow, 1, numRows, VIEWLOG_HEADERS.length).getValues();
    const rows = data.map(r => ({
      time:      r[0] instanceof Date ? Utilities.formatDate(r[0], 'Asia/Seoul', 'yyyy-MM-dd HH:mm:ss') : String(r[0] || ''),
      loginType: String(r[1] || ''),
      team:      String(r[2] || ''),
      userName:  String(r[3] || ''),
      ip:        String(r[4] || ''),
      page:      String(r[5] || '')
    }));
    return respond({ ok: true, rows });
  } catch(err) { return respond({ ok: false, msg: err.toString() }); }
}

// 3. doGet(e) 함수 안에 `if (action === 'getViewLogs') return getViewLogs();` 한 줄 추가.
